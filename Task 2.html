from collections import Counter
import re

# Split the hashtags column and flatten it into one list
hashtag_series = df['Hashtags'].str.lower().str.split()
flattened_hashtags = [
    tag for sublist in hashtag_series.dropna()
    for tag in sublist if re.match(r'#\w+', tag)
]

# Count the top 10 hashtags
hashtag_counts = Counter(flattened_hashtags).most_common(10)

# Print the results
print("Top 10 Hashtags:")
for tag, count in hashtag_counts:
    print(f"{tag} → {count} times")
